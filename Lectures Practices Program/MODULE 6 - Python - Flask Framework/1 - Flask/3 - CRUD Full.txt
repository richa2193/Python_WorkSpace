- def index():
    if request.method=='POST':		# check if the form has been submitted using the POST method.
        nm=request.form['name']	# gets the value entered in name field & stores in variable nm
        em=request.form['email']	# gets the value entered in email field & stores in variable em
        ct=request.form['city']		# same like name & email field

        stdata=Studinfo(name=nm,email=em,city=ct)	# create a new studinfo object with entered name, email & city
        db.session.add(stdata)	# adds the new project to the db session
        db.session.commit()		# save the record per. in db
    return render_template('index.html')	#display the html page to user

def deletedata(id):
	stid=Studinfo.query.get(id)
	db.session.delete(stid)
	db.session.commit()
	return redirect('/showdata')
