- from flask import Flask, render_templates, request

def index():
	if request.method=="POST":	# This check whether the request coming from the browser is a POST Request. 
		name=request.form.get("name")	#request.form containes all form data submitted by the user.
		city=request.form.get("city")		# same like name
		stdata=studinfo(name=name, city=city)		# create a new object of the studinfo model
		db.session.add(stdata)		# for adding record in db
		db.session.commit()		# save all pending changes in the session of the db
	return render_template('index.html')	# display the html page
